> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Expand BIN CITY from one shift into a five shift story across three districts with five visible ranks, living streets, interiors, a boss, cutscenes and a stylised visual overhaul.

**Architecture:** One Godot 4.7 project, procedural geometry and audio, a single `Game` orchestrator, a `GameState` autoload for progression and saves, area builders behind a shared `District` base, and a data-driven dialogue and quest layer. **Tech Stack:** Godot 4.7.2 (GL Compatibility, web export), GDScript, procedural shaders, Pillow studio renderer for the poster, pytest for the harness, Playwright and Chrome for playtests.

- [ ] **Step 4: Implement `game_state.gd`**

```gdscript
extends Node

signal rank_changed(rank: int)
signal xp_changed(xp: int, rank: int)
signal quest_changed(id: String)
signal story_changed(flag: String)

const SAVE_PATH := "user://bincity_save.cfg"

var rank := 1
var xp := 0
var shift_index := 1
var unlocked_shift := 1
var flags: Dictionary = {}
var quests: Dictionary = {}
var best_scores: Dictionary = {}
var settings := {"muted": false, "reduced_motion": false, "quality": "standard"}

func _ready() -> void:
	process_mode = Node.PROCESS_MODE_ALWAYS

func reset_new_game() -> void:
	rank = 1
	xp = 0
	shift_index = 1
	unlocked_shift = 1
	flags = {"met_nell": true}
	quests = {}
	best_scores = {}
	rank_changed.emit(rank)
	xp_changed.emit(xp, rank)

func xp_floor(value: int) -> int:
	var table: Array = load("res://scripts/data.gd").RANKS
	for entry in table:
		if int(entry.rank) == value:
			return int(entry.xp)
	return 0

func rank_for_xp(value: int) -> int:
	var result := 1
	for entry in load("res://scripts/data.gd").RANKS:
		if value >= int(entry.xp):
			result = int(entry.rank)
	return result

func add_xp(amount: int) -> bool:
	xp += maxi(0, amount)
	var new_rank := rank_for_xp(xp)
	var ranked := new_rank > rank
	rank = new_rank
	xp_changed.emit(xp, rank)
	if ranked:
		rank_changed.emit(rank)
		save()
	return ranked

func set_flag(name: String) -> void:
	flags[name] = true
	story_changed.emit(name)
	save()

func has_flag(name: String) -> bool:
	return bool(flags.get(name, false))

func start_quest(id: String) -> void:
	if quests.has(id):
		return
	var table: Dictionary = load("res://scripts/data.gd").QUESTS
	if not table.has(id):
		return
	quests[id] = {"status": "active", "progress": 0, "goal": int(table[id].goal)}
	quest_changed.emit(id)
	save()

func advance_quest(id: String, amount := 1) -> void:
	if quest_status(id) != "active":
		return
	var entry: Dictionary = quests[id]
	entry.progress = mini(int(entry.goal), int(entry.progress) + amount)
	if int(entry.progress) >= int(entry.goal):
		entry.status = "done"
		add_xp(int(load("res://scripts/data.gd").QUESTS[id].xp))
	quest_changed.emit(id)
	save()

func complete_quest(id: String) -> void:
	if quests.has(id):
		quests[id].status = "done"
		quest_changed.emit(id)
		save()

func quest_status(id: String) -> String:
	return str(quests.get(id, {}).get("status", "inactive"))

func save() -> void:
	var config := ConfigFile.new()
	config.load(SAVE_PATH)
	config.set_value("progress", "rank", rank)
	config.set_value("progress", "xp", xp)
	config.set_value("progress", "shift_index", shift_index)
	config.set_value("progress", "unlocked_shift", unlocked_shift)
	config.set_value("progress", "flags", flags)
	config.set_value("progress", "quests", quests)
	config.set_value("progress", "best_scores", best_scores)
	config.set_value("settings", "muted", settings.muted)
	config.set_value("settings", "reduced_motion", settings.reduced_motion)
	config.set_value("settings", "quality", settings.quality)
	config.save(SAVE_PATH)

func load_save() -> bool:
	var config := ConfigFile.new()
	if config.load(SAVE_PATH) != OK:
		return false
	rank = int(config.get_value("progress", "rank", 1))
	xp = int(config.get_value("progress", "xp", 0))
	shift_index = int(config.get_value("progress", "shift_index", 1))
	unlocked_shift = int(config.get_value("progress", "unlocked_shift", 1))
	flags = config.get_value("progress", "flags", {})
	quests = config.get_value("progress", "quests", {})
	best_scores = config.get_value("progress", "best_scores", {})
	settings.muted = bool(config.get_value("settings", "muted", false))
	settings.reduced_motion = bool(config.get_value("settings", "reduced_motion", false))
	settings.quality = str(config.get_value("settings", "quality", "standard"))
	rank_changed.emit(rank)
	xp_changed.emit(xp, rank)
	return true

func has_save() -> bool:
	return FileAccess.file_exists(SAVE_PATH)
```

Note: dictionary dot access (`settings.muted`) works for keys in GDScript 4. Keep values serialisable.

- [ ] **Step 2: Implement the post shader**

```glsl
shader_type canvas_item;

uniform sampler2D screen_texture : hint_screen_texture, repeat_disable, filter_nearest;
uniform vec3 warm : source_color = vec3(1.0, 0.93, 0.82);
uniform vec3 cool : source_color = vec3(0.72, 0.85, 0.92);
uniform float strength : hint_range(0.0, 1.0) = 0.55;
uniform float vignette : hint_range(0.0, 1.2) = 0.35;
uniform float grain : hint_range(0.0, 0.2) = 0.035;
uniform float time = 0.0;

float hash(vec2 p) {
	return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453);
}

void fragment() {
	vec3 base = textureLod(screen_texture, SCREEN_UV, 0.0).rgb;
	vec3 col = base;
	float luma = dot(col, vec3(0.299, 0.587, 0.114));
	col *= mix(cool, warm, smoothstep(0.15, 0.85, luma));
	col = mix(col, vec3(luma), 0.08);
	col = mix(col, col * col * (3.0 - 2.0 * col), 0.12);
	float d = distance(SCREEN_UV, vec2(0.5));
	col *= 1.0 - vignette * smoothstep(0.35, 0.85, d);
	col += (hash(SCREEN_UV * 1024.0 + vec2(time)) - 0.5) * grain;
	COLOR = vec4(mix(base, col, strength), 1.0);
}
```

The final mix with `strength` keeps the grade subtle. Update `time` from
`postfx.gd` each frame, but freeze it under reduced motion. Use a nearest
filter and `textureLod`; mipmapped screen reads are slow on weak GPUs and
broken on some web browsers. Do not add a `BackBufferCopy`; a single
CanvasLayer screen read copies the frame automatically. Do not rely on
`use_debanding`, SSAO, volumetric fog or DOF in the web build, because
Compatibility does not provide them.

```gdscript
	_check("boss starts at full", boss.health == 300.0)
	boss.force_open("paper")
	var result = boss.feed("paper", [{"kind": "paper", "weight": 1.0}])
	_check("correct feed damages", result.damage > 0.0)
	_check("wrong chute rejected", boss.feed("glass", [{"kind": "paper"}]).damage == 0.0)
	boss.health = 200.0
	boss._check_phase()
	_check("phase 2 at 200", boss.phase == 2)
	boss.queue_free()
```

- [ ] **Step 2: Implement the boss state machine**

Three phases at 300, 200 and 100 health. `_process` runs the patrol, suction, junk volleys, chute windows and telegraphs. `force_open(kind)` is debug and also used by the scheduler. `feed(kind, items)` consumes matching items, deals 8 damage each, rejects wrong kinds, and triggers a stomp every third hit. Phase 2 adds grime patches and two open chutes. Phase 3 stops damage and opens the core for the hopper unload.

- [ ] **Step 3: Build the arena**

A circular plant floor with a conveyor ring, three chutes on the Gulper's flank, steam vents, warning lamps, a raised viewing gantry for Pip and Rook, and a safe zone at the entrance. The Gulper is a large machine: a drum body, intake mouth with teeth, three chute doors, exhaust stacks, treads and a single warm eye.

- [ ] **Step 4: Wire the fight**

When the player presses interact near an open chute with matching items in the hopper, consume them with a toss animation, deal damage and play the chute sound. Wrong items bounce out with a warning. Update the boss bar. At zero on phase 3, play the ending cutscene and credits.

- [ ] **Step 5: Telegraphed hazards**

Suction: a visible cone and a pull on the player. Stomp: a ring shockwave that drops the stack unless jumped. Junk volley: arcing lumps that can be vacuumed into the hopper. All telegraphs survive reduced motion with no shake.
